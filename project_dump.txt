<span class="line-highlight">2. Top 3 products account for 60% of sales...</span></code></pre>
              </div>

              <p class="doc-text section-note">This prompt scores <strong>9.2/10</strong> because it includes a role, format, constraint, tone, and one example — all six criteria are satisfied.</p>

            clarity:  text.length > 50
          };

          TASK.criteria.forEach(function(c) {
            var icon = document.getElementById('icon-' + c.key);
            var item = document.querySelector('[data-criterion="' + c.key + '"]');
            if (checks[c.key]) {
              icon.textContent = '✓';
              icon.classList.add('criteria-met');
              item.classList.add('criteria-met');
            } else {
              icon.textContent = '○';
              icon.classList.remove('criteria-met');
              item.classList.remove('criteria-met');

          });
        }

        // SUBMIT — mock scoring
        // =========================================
        submitBtn.addEventListener('click', function() {
          var text = promptInput.value.trim();
          if (text.length < 10) {
            promptInput.focus();
            promptInput.classList.add('editor-error');
            setTimeout(function() { promptInput.classList.remove('editor-error'); }, 1500);
            return;
          }

          // Simulate evaluation (will be replaced by server call)
          var scores = {};
          var total = 0;
          TASK.criteria.forEach(function(c) {
            var s = Math.random() * 4 + 6; // 6–10 range for demo
            s = Math.round(s * 10) / 10;
            scores[c.key] = s;
            total += s * c.weight;
          });
          total = Math.round(total * 100) / 100;

          displayResults(scores, total);
          saveAttempt(text, scores, total);

        // DISPLAY RESULTS

        function displayResults(scores, total) {
          resultsPanel.hidden = false;

          resultsPanel.scrollIntoView({ behavior: 'smooth', block: 'start' });

        // =========================================

        // =========================================
        function saveAttempt(text, scores, total) {
          var attempt = {
            id: attempts.length + 1,
            text: text,
            scores: scores,
            total: total,
            time: new Date().toLocaleTimeString()
          };
          attempts.push(attempt);
          renderHistory();
        }

        // =========================================
        // RENDER HISTORY
        // =========================================
        function renderHistory() {
          if (attempts.length === 0) {
            historyEmpty.hidden = false;
            historyList.hidden = true;
            return;
          }
          historyEmpty.hidden = true;
          historyList.hidden = false;
          historyList.innerHTML = '';

          // render in reverse order (newest first)
          for (var i = attempts.length - 1; i >= 0; i--) {
            var a = attempts[i];
            var li = document.createElement('li');
            li.className = 'history-item';
            li.innerHTML =
              '<div class="history-item-header">' +
                '<span class="history-attempt-num mono">Attempt #' + a.id + '</span>' +
                '<span class="history-time">' + a.time + '</span>' +
              '</div>' +
              '<div class="history-score">' +
                '<span class="history-final mono">' + a.total.toFixed(2) + '</span>' +
                '<span class="history-max">/ 10</span>' +
              '</div>' +
              '<div class="history-preview">' + escapeHtml(a.text.substring(0, 120)) + (a.text.length > 120 ? '…' : '') + '</div>';
            historyList.appendChild(li);
          }
        }

# PromptForge — полное техническое задание
### Assignment 1 (HTML) + Assignment 2 (CSS) + задел под будущий движок оценки (JS/backend)

Команда: 3 человека. Центральное правило: **после Assignment 1 новые .html файлы не создаются** — вся карта страниц фиксируется в этом документе один раз, дальше только наполняется и красится.

## 0. Правила, действующие на КАЖДОЙ из 10 страниц

- `<!DOCTYPE html>`, `lang="en"`, `charset`, `viewport`, `<meta name="description">`, `<meta name="author" content="Имя">`, уникальный `<title>`.
- Ровно один `<h1>`, без пропуска уровней заголовков.
- Скелет: `<header>` → `<nav>` со ссылками на все 10 страниц → `<main>` → `<footer>` (контакты, `&copy;`).
- HTML-комментарий с именем автора + минимум 2 комментария «почему» (не «что»).
- Никакого `style=`, `<style>`, ссылок на CSS, `<script>` на этом этапе.

**Общий `<nav>` (копируется вручную в каждый файл):**
Главная · Задачи · Как это работает · Рейтинг · Мои решения · Профиль · Регистрация · Вход · Colophon

## 2. Ключевое решение: `task.html` — шаблон, а не страница одной задачи

Раз новых HTML-файлов больше не будет, отдельных `task-1.html`, `task-2.html`... быть не может.

- `task.html` на этапе HTML показывает один конкретный пример целиком (условие, критерии, форма, готовая оценка) — так закрываются все требуемые теги.
- `tasks.html` — каталог, карточки задач ссылаются на `task.html` (пока все — на один и тот же файл).
- На этапе JS: `task.html` читает номер задачи из URL (`task.html?id=3`) или `localStorage` и подставляет контент — файл не меняется, меняется то, что в него подставляет скрипт.
- В `task.html` держите «условие + форма» и «результат оценки» в двух разных `<section>` — потом вторую секцию будет легко скрывать/показывать скриптом.
- Осмысленные `name`-атрибуты у полей форм (`name="user_prompt"`, `name="task_id"`) — JS на будущем этапе будет читать форму по именам, а не по случайным id.
- **Таблица критериев на `task.html` — это не просто вывод, а будущая схема рубрики**: каждая строка таблицы («Критерий | Вес | Балл») один в один станет одним атомарным критерием оценщика на этапе JS/бэкенда (см. раздел 9). Поэтому таблицу стоит проектировать сразу с прицелом на 4-6 отдельных, независимых друг от друга критериев, а не на общую фразу «оцените промпт».

## 9. Движок оценки промптов — задел под этап JS/backend

Это уже не HTML/CSS, но форма и таблица критериев на `task.html` должны быть готовы к этой логике без переделки разметки.

**Проблема 1 — один эталон наказывает валидные альтернативы.**
Не сравнивать промпт целиком с эталонным текстом, а разбивать оценку на атомарные независимые критерии («есть ли явная роль для модели — да/нет», «указан ли формат вывода — да/нет», «есть ли ограничение по длине — да/нет»). Таблица критериев на `task.html` уже сейчас должна быть спроектирована как список таких отдельных пунктов, а не как одна общая фраза «оцените промпт». Плюс держать 2–3 эталонных примера разных валидных стратегий на задачу, а не один-единственный «правильный» промпт — оцениваются они все против одной и той же рубрики, а не друг против друга.

**Проблема 2 — нестабильность ИИ-судьи.**
- Атомарные да/нет-критерии сами по себе снижают шум по сравнению с одним общим баллом.
- Оценка с опорой на эталонный пример надёжнее, чем судья «с нуля».
- Несколько прогонов судьи (2–3 раза) с медианой по результату заметно повышают стабильность; после 3–4 прогонов прирост стабильности резко падает, гонять 10 раз не имеет смысла.
- Панель из разных моделей-судей повышает согласованность сильнее, чем повторные прогоны одной и той же модели.
- Нужен калибровочный набор: 30–50 промптов, оценённых вручную (вами/куратором), сверка с судьёй — если расхождение выше 20–25%, рубрику нужно переписать, а не доверять цифре как есть.

**Проблема 3 — привязка к конкретной модели.**
Явно фиксировать на `task.html`, под какую модель заточена задача («проверяется на GPT-4/Claude»). В идеале — реально прогонять промпт пользователя через целевую модель и оценивать полученный ответ, а не только текст промпта, и показывать ответ модели рядом с ответом на эталон.

**Что это значит для разметки уже сейчас:** таблица критериев = будущая схема рубрики (одна строка = один атомарный чек), блок итогового балла — изолированный `span`, чтобы JS мог подставлять туда агрегированное (медианное) значение без правки структуры, а не единичное число от одного вызова.

import os
from pathlib import Path

PROJECT_DIR = Path(__file__).parent
OUTPUT_FILE = PROJECT_DIR / "project_dump.txt"

INCLUDE_EXTENSIONS = {".html", ".css", ".js", ".md", ".json", ".py", ".svg"}
EXCLUDE_DIRS = {".git", ".vscode", "node_modules", "__pycache__"}
EXCLUDE_FILES = {"project_dump.txt", ".gitignore"}


def collect_files():
    files = []
    for root, dirs, filenames in os.walk(PROJECT_DIR):
        dirs[:] = [d for d in dirs if d not in EXCLUDE_DIRS]
        for name in sorted(filenames):
            if name in EXCLUDE_FILES:
                continue
            path = Path(root) / name
            if path.suffix in INCLUDE_EXTENSIONS:
                files.append(path)
    return sorted(files, key=lambda p: (p.suffix, str(p.relative_to(PROJECT_DIR))))


def build_dump(files):
    lines = []
    lines.append(f"{'=' * 60}")
    lines.append(f"PromptForge — project source dump")
    lines.append(f"Files: {len(files)}")
    lines.append(f"{'=' * 60}")
    lines.append("")

    for f in files:
        rel = f.relative_to(PROJECT_DIR)
        try:
            content = f.read_text(encoding="utf-8")
        except Exception as e:
            content = f"[ERROR reading file: {e}]"

        lines.append(f"{'─' * 60}")
        lines.append(f"FILE: {rel}")
        lines.append(f"{'─' * 60}")
        lines.append(content)
        lines.append("")

    return "\n".join(lines)


def main():
    files = collect_files()
    dump = build_dump(files)
    OUTPUT_FILE.write_text(dump, encoding="utf-8")
    print(f"Dumped {len(files)} files → {OUTPUT_FILE.name}")


if __name__ == "__main__":
    main()
